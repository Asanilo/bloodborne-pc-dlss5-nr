cmake_minimum_required(VERSION 3.20)
project(bbport_resource_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
add_executable(nr-probe nr_probe.cpp)
target_link_libraries(nr-probe PRIVATE Vulkan::Vulkan)
target_compile_options(nr-probe PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
add_executable(dlss-sr-smoke dlss_sr_smoke.cpp)
target_link_libraries(dlss-sr-smoke PRIVATE Vulkan::Vulkan ${CMAKE_DL_LIBS})
target_compile_options(dlss-sr-smoke PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
